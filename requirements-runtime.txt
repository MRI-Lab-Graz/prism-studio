# Direct floor, not just a transitive requests pin: requests itself only
# requires urllib3>=1.26. GHSA-38jv-5279-wg99 / CVE-2026-21441 (decompression
# bomb via redirect handling in the streaming API) is fixed in 2.6.3; pinning
# here stops a resolver from settling on an older, vulnerable urllib3.
urllib3>=2.8.0
# The BIDS check runs by default and needs this engine; the PyPI package `deno` it depends on
# ships the Deno runtime as a wheel for these platforms only, so the requirement is skipped on other
# architectures (e.g. Windows ARM64: install succeeds, a run reports PRISM902 and the --no-bids hint).
# On musl and glibc < 2.27 the marker matches but no wheel exists and the sdist build fails: the
# install fails there (use the Docker image or a glibc >= 2.27 system).
bids-validator-deno>=3.0.2,<4; sys_platform == "darwin" or (sys_platform == "linux" and platform_machine in "x86_64 aarch64") or (sys_platform == "win32" and platform_machine == "AMD64")
pandas
openpyxl
waitress
pyreadstat
pyreadr
defusedxml
# pywebview provides the native app window on macOS (Cocoa/WebKit via pyobjc).
# Windows/Linux use a Chromium app-mode window instead (see app/prism-studio.py),
# so pywebview — and its heavy pythonnet/.NET stack on Windows — is not needed
# there and is intentionally scoped to macOS only.
pywebview; sys_platform == "darwin"
# cryptography 49+ dropped Intel macOS wheels (arm64-only). A flat version
# pin here gets silently re-widened by dependabot (it can't read this
# comment) and has broken the Intel macOS build twice (2026-07-22,
# 2026-08-08). Split by platform instead: Intel macOS stays on the last
# release with an Intel wheel (48.0.1); everywhere else tracks latest.
cryptography>=50.0.0,<51; platform_machine != "x86_64" or sys_platform != "darwin"
cryptography>=46.0.7,<49; platform_machine == "x86_64" and sys_platform == "darwin"
